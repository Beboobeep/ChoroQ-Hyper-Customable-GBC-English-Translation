%%% to calculate pointer addresses ::  %%%%%%%%%%%%%%%%%%%%%


( 0x< ROM Address > ) / ( 0x4000 ) = ( 0x < BANK# > ) 

( 0x < BANK# > ) * ( 0x4000 ) = ( 0x <BANK Start Address )

( 0x< ROM Address > ) - ( 0x <BANK Start Address ) = ( 0x < Pointer Address NOT Accounting for BANK 0 > )

( 0x < Pointer Address NOT Accounting for BANK 0 > ) + ( 0x4000) = ( 0x < Pointer Address )


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

ROM file offset = (bank_number * 0x4000) + (CPU address - 0x4000)

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
